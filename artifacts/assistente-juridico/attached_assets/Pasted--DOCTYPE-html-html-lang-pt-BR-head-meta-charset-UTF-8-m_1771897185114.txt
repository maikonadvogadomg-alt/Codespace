// ====== FUNÇÕES AUXILIARES ======
function id(x) { return document.getElementById(x); }

function escapeHtml(text) {
  const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
  return text.replace(/[&<>"']/g, m => map[m]);
}

function nl2br(text) {
  return escapeHtml(text).replace(/\n/g, '<br>');
}

// ====== ATUALIZAR DOCUMENTO ======
function updateDoc() {
  // Tipo de peça
  const tipo = id('tipoPeca').value;
  const config = tiposPeca[tipo];
  
  id('v_nomePeca').innerText = config.nome;
  id('v_tipoParte').innerText = config.parte;
  id('v_tipoParteContraria').innerText = config.parteContraria;

  // Campos básicos
  id('v_destinatario').innerText = id('destinatario').value || 'EXCELENTÍSSIMO SENHOR DOUTOR JUIZ FEDERAL';
  id('v_processo').innerText = id('processo').value || '[Nº do processo]';
  id('v_parte').innerText = id('parte').value || '[Nome da parte]';
  id('v_parte2').innerText = id('parte').value || '[Nome da parte]';
  id('v_qualificacao').innerText = id('qualificacao').value || '[qualificação completa]';
  id('v_parteContraria').innerText = id('parteContraria').value || '[Parte contrária]';
  
  // Conteúdo
  id('v_sintese').innerHTML = nl2br(id('sintese').value) || '<em>[Descreva a síntese fática]</em>';
  id('v_fundamento').innerHTML = nl2br(id('fundamento').value) || '<em>[Desenvolva a fundamentação jurídica]</em>';
  id('v_pedidos').innerHTML = nl2br(id('pedidos').value) || '<em>[Liste os pedidos]</em>';
  
  // Assinatura
  id('v_localData').innerText = id('localData').value || '[Local e data]';
  id('v_advogado').innerText = id('advogado').value || '[Advogado - OAB]';

  // Tópicos extras
  renderTopicosExtras();
}

// ====== ADICIONAR TÓPICO EXTRA ======
function adicionarTopico() {
  contadorTopicos++;
  const topicoId = `topico_${contadorTopicos}`;
  
  topicos.push({
    id: topicoId,
    numero: contadorTopicos
  });

  const container = id('topicosExtras');
  const div = document.createElement('div');
  div.className = 'topico-extra';
  div.id = topicoId;
  
  div.innerHTML = `
    <div class="topico-extra-header">
      <strong>Tópico ${contadorTopicos}</strong>
      <button class="btn-remove" onclick="removerTopico('${topicoId}')" title="Remover tópico">×</button>
    </div>
    
    <label>Tipo de conteúdo:</label>
    <select id="${topicoId}_tipo" onchange="updateDoc()">
      <option value="subtitulo">Subtítulo (h3)</option>
      <option value="paragrafo">Parágrafo normal</option>
      <option value="citacao">Citação jurídica (4cm)</option>
    </select>
    
    <label>Título/Subtítulo:</label>
    <input type="text" id="${topicoId}_titulo" placeholder="Ex: 2.1. Da omissão quanto à DIB" oninput="updateDoc()">
    
    <label>Conteúdo:</label>
    <textarea id="${topicoId}_conteudo" oninput="updateDoc()" style="min-height: 100px;" 
              placeholder="Digite o conteúdo deste tópico..."></textarea>
  `;
  
  container.appendChild(div);
  updateDoc();
}

// ====== REMOVER TÓPICO ======
function removerTopico(topicoId) {
  const index = topicos.findIndex(t => t.id === topicoId);
  if (index > -1) {
    topicos.splice(index, 1);
  }
  
  const elemento = id(topicoId);
  if (elemento) {
    elemento.remove();
  }
  
  updateDoc();
}

// ====== RENDERIZAR TÓPICOS EXTRAS NO PREVIEW ======
function renderTopicosExtras() {
  const container = id('v_topicosExtras');
  let html = '';
  
  topicos.forEach(topico => {
    const tipo = id(`${topico.id}_tipo`)?.value || 'paragrafo';
    const titulo = id(`${topico.id}_titulo`)?.value || '';
    const conteudo = id(`${topico.id}_conteudo`)?.value || '';
    
    if (!titulo && !conteudo) return;
    
    if (tipo === 'subtitulo') {
      html += `<h3>${escapeHtml(titulo)}</h3>`;
      if (conteudo) {
        html += `<p>${nl2br(conteudo)}</p>`;
      }
    } else if (tipo === 'citacao') {
      if (titulo) {
        html += `<h3>${escapeHtml(titulo)}</h3>`;
      }
      html += `<div class="citacao">${nl2br(conteudo)}</div>`;
    } else {
      if (titulo) {
        html += `<h3>${escapeHtml(titulo)}</h3>`;
      }
      html += `<p>${nl2br(conteudo)}</p>`;
    }
  });
  
  container.innerHTML = html;
}

// ====== EXPORTAR PARA WORD ======
async function exportarWord() {
  const { Document, Packer, Paragraph, TextRun, HeadingLevel, AlignmentType } = docx;
  
  const children = [];
  
  // Cabeçalho
  children.push(
    new Paragraph({
      text: id('v_destinatario').innerText,
      alignment: AlignmentType.CENTER,
      spacing: { after: 200 }
    }),
    new Paragraph({
      children: [
        new TextRun({ text: 'PROCESSO Nº: ', bold: true }),
        new TextRun(id('v_processo').innerText)
      ],
      spacing: { after: 100 }
    }),
    new Paragraph({
      children: [
        new TextRun({ text: id('v_tipoParte').innerText + ': ', bold: true }),
        new TextRun(id('v_parte').innerText)
      ],
      spacing: { after: 100 }
    }),
    new Paragraph({
      children: [
        new TextRun({ text: id('v_tipoParteContraria').innerText + ': ', bold: true }),
        new TextRun(id('v_parteContraria').innerText)
      ],
      spacing: { after: 200 }
    })
  );
  
  // Preâmbulo
  const preambulo = `${id('parte').value}, ${id('qualificacao').value}, por intermédio de seu advogado que esta subscreve, vem, respeitosamente, à presença de Vossa Excelência, apresentar ${id('v_nomePeca').innerText}, com fundamento no art. 1.022 do Código de Processo Civil, pelos motivos de fato e de direito a seguir expostos.`;
  
  children.push(
    new Paragraph({
      text: preambulo,
      alignment: AlignmentType.JUSTIFIED,
      indent: { firstLine: 1134 },
      spacing: { after: 200 }
    })
  );
  
  // Síntese Fática
  children.push(
    new Paragraph({
      text: 'I – SÍNTESE FÁTICA',
      heading: HeadingLevel.HEADING_2,
      spacing: { before: 200, after: 100 }
    }),
    new Paragraph({
      text: id('sintese').value,
      alignment: AlignmentType.JUSTIFIED,
      indent: { firstLine: 1134 },
      spacing: { after: 200 }
    })
  );
  
  // Fundamentação
  children.push(
    new Paragraph({
      text: 'II – FUNDAMENTAÇÃO JURÍDICA',
      heading: HeadingLevel.HEADING_2,
      spacing: { before: 200, after: 100 }
    }),
    new Paragraph({
      text: id('fundamento').value,
      alignment: AlignmentType.JUSTIFIED,
      indent: { firstLine: 1134 },
      spacing: { after: 200 }
    })
  );
  
  // Tópicos extras
  topicos.forEach(topico => {
    const tipo = id(`${topico.id}_tipo`)?.value;
    const titulo = id(`${topico.id}_titulo`)?.value;
    const conteudo = id(`${topico.id}_conteudo`)?.value;
    
    if (titulo) {
      children.push(
        new Paragraph({
          text: titulo,
          heading: HeadingLevel.HEADING_3,
          spacing: { before: 150, after: 100 }
        })
      );
    }
    
    if (conteudo) {
      const indent = tipo === 'citacao' ? { left: 2268 } : { firstLine: 1134 };
      children.push(
        new Paragraph({
          text: conteudo,
          alignment: AlignmentType.JUSTIFIED,
          indent: indent,
          spacing: { after: 200 }
        })
      );
    }
  });
  
  // Pedidos
  children.push(
    new Paragraph({
      text: 'III – PEDIDOS',
      heading: HeadingLevel.HEADING_2,
      spacing: { before: 200, after: 100 }
    }),
    new Paragraph({
      text: id('pedidos').value,
      alignment: AlignmentType.JUSTIFIED,
      indent: { firstLine: 1134 },
      spacing: { after: 200 }
    })
  );
  
  // Encerramento
  children.push(
    new Paragraph({
      text: 'Termos em que,',
      alignment: AlignmentType.CENTER,
      spacing: { before: 300, after: 100 }
    }),
    new Paragraph({
      text: 'Pede deferimento.',
      alignment: AlignmentType.CENTER,
      spacing: { after: 300 }
    }),
    new Paragraph({
      text: id('localData').value,
      alignment: AlignmentType.CENTER,
      spacing: { after: 200 }
    }),
    new Paragraph({
      text: id('advogado').value,
      alignment: AlignmentType.CENTER,
      bold: true
    })
  );
  
  const doc = new Document({
    sections: [{
      properties: {
        page: {
          margin: {
            top: 1134,
            right: 1134,
            bottom: 1134,
            left: 1134
          }
        }
      },
      children: children
    }]
  });
  
  const blob = await Packer.toBlob(doc);
  saveAs(blob, `${id('tipoPeca').value}_${id('processo').value || 'documento'}.docx`);
}

// ====== LIMPAR TUDO ======
function limparTudo() {
  if (!confirm('Tem certeza que deseja limpar todos os campos?')) return;
  
  id('destinatario').value = '';
  id('processo').value = '';
  id('parte').value = '';
  id('qualificacao').value = '';
  id('parteContraria').value = '';
  id('sintese').value = '';
  id('fundamento').value = '';
  id('pedidos').value = '';
  id('localData').value = '';
  id('advogado').value = '';
  
  topicos.length = 0;
  contadorTopicos = 0;
  id('topicosExtras').innerHTML = '';
